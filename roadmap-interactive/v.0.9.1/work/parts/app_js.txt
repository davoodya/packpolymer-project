  const DIGIT_MAP = { '\u06f0':'0','\u06f1':'1','\u06f2':'2','\u06f3':'3','\u06f4':'4','\u06f5':'5','\u06f6':'6','\u06f7':'7','\u06f8':'8','\u06f9':'9',

  function norm(s) {
    return String(s == null ? '' : s).toLowerCase()
      .replace(/[\u06f0-\u06f9\u0660-\u0669]/g, (d) => DIGIT_MAP[d])
      .replace(/[\u064a\u0649]/g, '\u06cc')      // ي / ى → ی

      .replace(/\u0629/g, '\u0647')                // ة → ه
      .replace(/[\u064b-\u0652\u0640]/g, '')      // اعراب و کشیده
      .replace(/[\u200c\u200d\u200e\u200f]/g, ' ') // کنترل های قدیمی را به فاصله معمولی تبدیل کن
      .replace(/[\u00a0\t\r\n]+/g, ' ')
      .replace(/\s{2,}/g, ' ')
      .trim();

    function open() {
      return new Promise((res) => {
        try {
          if (!window.indexedDB) return res(null);
          const rq = indexedDB.open('packpolymer-roadmap', 1);
          rq.onupgradeneeded = () => { try { rq.result.createObjectStore('kv'); } catch (e) { } };
          rq.onsuccess = () => res(rq.result);
          rq.onerror = () => res(null);
        } catch (e) { res(null); }
      });
    }
    return {
      async set(k, v) {
        const d = await open(); if (!d) return false;
        return new Promise((res) => {
          try {
            const tx = d.transaction('kv', 'readwrite'); tx.objectStore('kv').put(v, k);
            tx.oncomplete = () => res(true); tx.onerror = () => res(false);
          } catch (e) { res(false); }
        });
      },
      async get(k) {
        const d = await open(); if (!d) return null;
        return new Promise((res) => {
          try {
            const tx = d.transaction('kv', 'readonly'); const rq = tx.objectStore('kv').get(k);
            rq.onsuccess = () => res(rq.result || null); rq.onerror = () => res(null);
          } catch (e) { res(null); }
        });
      },
      async del(k) {
        const d = await open(); if (!d) return false;
        return new Promise((res) => {
          try {
            const tx = d.transaction('kv', 'readwrite'); tx.objectStore('kv').delete(k);
            tx.oncomplete = () => res(true); tx.onerror = () => res(false);
          } catch (e) { res(false); }
        });
      }
    };

        try { perm = await h.queryPermission({ mode: 'readwrite' }); } catch (e) { perm = 'prompt'; }

      } catch (e) { }

  function answersEmpty(a) {
    if (!a) return true;
    const hasChecks = a.checks && Object.keys(a.checks).length > 0;
    const hasNotes = a.notes && Object.keys(a.notes).length > 0;
    const hasMarks = a.marks && ((a.marks.archived && Object.keys(a.marks.archived).length) || (a.marks.review && Object.keys(a.marks.review).length));
    const hasNb = (a.notebook && String(a.notebook).trim().length > 0) ||
      (Array.isArray(a.nbNotes) && a.nbNotes.some(n => n && String(n.body || '').trim().length > 0));
    return !hasChecks && !hasNotes && !hasNb && !hasMarks;
  }

  /* ---------- دفترچه چندیادداشتی: مدل داده --------------------------------
   * منبع اصلی: state.nbNotes = [{id,title,body,createdAt,updatedAt}]
   * state.nbActiveId آخرین یادداشت بازشده است تا با بازکردن دوباره دفترچه
   * دقیقاً همان یادداشت (Last Note) ادامه پیدا کند.
   * state.notebook فقط برای سازگاری با فرمت قدیمی/گزارش نگه داشته می شود.

  function nbNormalizeNote(n) {
    if (!n || typeof n !== 'object') return null;
    const now = new Date().toISOString();
    return {
      id: n.id || uid('nb'),
      title: typeof n.title === 'string' ? n.title : '',
      body: typeof n.body === 'string' ? n.body : '',
      createdAt: n.createdAt || now,
      updatedAt: n.updatedAt || n.createdAt || now
    };
  }
  function nbTitleFromBody(body) {
    const lines = String(body || '').split('\n').map(x => x.trim()).filter(Boolean);
    if (!lines.length) return 'یادداشت بدون عنوان';
    const h = lines[0].replace(/^#{1,6}\s*/, '').replace(/[*_`>]/g, '').trim();
    return (h || 'یادداشت بدون عنوان').slice(0, 60);
  }
  function nbAllText() {
    const list = (state.nbNotes || []).filter(n => n && (String(n.title || '').trim() || String(n.body || '').trim()));
    if (!list.length) return state.notebook && !(state.nbNotes && state.nbNotes.length) ? state.notebook : '';
    return list.map(n => '# ' + (String(n.title || '').trim() || nbTitleFromBody(n.body)) + '\n\n' + String(n.body || '').trim()).join('\n\n---\n\n');
  }
  function nbSyncLegacy() { state.notebook = nbAllText(); }
  function nbMigrate() {
    if (!Array.isArray(state.nbNotes)) state.nbNotes = [];
    state.nbNotes = state.nbNotes.map(nbNormalizeNote).filter(Boolean);
    // مهاجرت از فرمت تک یادداشتی قدیمی
    if (!state.nbNotes.length && typeof state.notebook === 'string' && state.notebook.trim()) {
      const now = new Date().toISOString();
      state.nbNotes = [{ id: uid('nb'), title: nbTitleFromBody(state.notebook), body: state.notebook, createdAt: now, updatedAt: now }];
    }
    if (state.nbNotes.length) {
      const ok = state.nbNotes.some(n => n.id === state.nbActiveId);
      if (!ok) {
        const sorted = state.nbNotes.slice().sort((a, b) => String(b.updatedAt).localeCompare(String(a.updatedAt)));
        state.nbActiveId = sorted[0].id;
      }
    } else state.nbActiveId = '';
    if (!state.nbNotes.some(n => n.id === state.nbSecondaryId) || state.nbSecondaryId === state.nbActiveId) {
      const other = state.nbNotes.find(n => n.id !== state.nbActiveId);
      state.nbSecondaryId = other ? other.id : '';
    }
    state.nbSplit = !!state.nbSplit && !!state.nbSecondaryId;
    if (state.nbMode !== 'read') state.nbMode = 'edit';
    state.nbSecondaryMode = state.nbSecondaryMode === 'edit' ? 'edit' : 'read';
    if (state.nbView !== 'note') state.nbView = 'list';
    if (state.nbView === 'note' && !state.nbActiveId) state.nbView = 'list';
    nbSyncLegacy();
  }
  function nbActive(side) {
    const id = side === 'secondary' ? state.nbSecondaryId : state.nbActiveId;
    return (state.nbNotes || []).find(n => n.id === id) || null;
  }
  function nbSorted(list) {
    return (list || state.nbNotes || []).slice().sort((a, b) => String(b.updatedAt || '').localeCompare(String(a.updatedAt || '')));

  /* اعمال «پاسخ ها» روی وضعیت فعلی (بدون دست زدن به ساختار مراحل پیش فرض) */
  function applyAnswers(a, silent) {
    if (!a) return false;
    state.checks = a.checks || {};
    state.notes = a.notes || {};
    if (a.marks) state.marks = { archived: a.marks.archived || {}, review: a.marks.review || {} };
    state.ui.sidebarLayout = validSidebarLayout(a.sidebarLayout);
    if (a.uiFontScale != null) state.ui.fontScale = clampScale(a.uiFontScale, state.ui.fontScale);
    if (a.nbFontScale != null) state.nbFontScale = clampScale(a.nbFontScale, state.nbFontScale);
    if (Array.isArray(a.nbNotes) && a.nbNotes.length) {
      state.nbNotes = a.nbNotes.map(nbNormalizeNote).filter(Boolean);
      state.nbActiveId = a.nbActiveId || '';
      state.nbSecondaryId = a.nbSecondaryId || '';
      state.nbSplit = !!a.nbSplit;
      state.nbMode = a.nbMode === 'read' ? 'read' : 'edit';
      state.nbSecondaryMode = a.nbSecondaryMode === 'edit' ? 'edit' : 'read';
      state.nbView = a.nbView || state.nbView || 'list';
      nbMigrate();
    } else if (typeof a.notebook === 'string') { state.notebook = a.notebook; nbMigrate(); }
    if (a.customized !== undefined) state.customized = !!a.customized;
    if (a.meta && a.meta.title && !state.customized) state.meta = Object.assign(state.meta, a.meta);
    if (a.savedAt) state.savedAt = a.savedAt;
    if (!silent) render();
    return true;
  }

          meta: s.meta || '',
          tip: s.tip || '',
          stack: s.stack || '',
          subs: (s.subs || []).map((x, i) => {
            if (typeof x === 'string') return { id: sid + '-' + (i + 1), title: x };
            return { id: x.id || (sid + '-' + (i + 1)), title: x.title || '' };
          })
        };
      })
    };
    return ph;
  }

      const d = JSON.parse(raw);
      if (!d || typeof d !== 'object') return;
      if (d.meta) state.meta = Object.assign(state.meta, d.meta);
      if (Array.isArray(d.plan) && d.plan.length) state.plan = d.plan.map(normalizePhase);
      state.checks = d.checks || {};
      state.notes = d.notes || {};
      if (d.marks) state.marks = { archived: d.marks.archived || {}, review: d.marks.review || {} };
      state.open = d.open || {};
      state.openNotes = d.openNotes || {};
      state.notebook = typeof d.notebook === 'string' ? d.notebook : '';
      if (Array.isArray(d.nbNotes)) state.nbNotes = d.nbNotes.map(nbNormalizeNote).filter(Boolean);
      state.nbActiveId = d.nbActiveId || '';
      state.nbSecondaryId = d.nbSecondaryId || '';
      state.nbSplit = !!d.nbSplit;
      state.nbMode = d.nbMode === 'read' ? 'read' : 'edit';
      state.nbSecondaryMode = d.nbSecondaryMode === 'edit' ? 'edit' : 'read';
      state.nbView = d.nbView || 'list';
      state.nbQuery = d.nbQuery || '';
      state.nbFontScale = clampScale(d.nbFontScale, 1);
      state.ui.fontScale = clampScale(d.ui && d.ui.fontScale, 1);
      state.ui.sidebarLayout = validSidebarLayout(d.ui && d.ui.sidebarLayout);
      state.ui.phaseDrawerOpen = false;
      state.savedAt = d.savedAt || '';
      state.ui.editing = false;
      if (d.ui) {
        state.ui.nbOpen = !!d.ui.nbOpen; state.ui.nbPreview = d.ui.nbPreview !== false; state.ui.reportMode = d.ui.reportMode || 'full';
        state.ui.phaseNavOpen = d.ui.phaseNavOpen !== false;
        state.ui.phaseNavScroll = Number(d.ui.phaseNavScroll) || 0;
        state.ui.sidebarLayout = validSidebarLayout(d.ui.sidebarLayout);
        state.ui.phaseDrawerQuery = typeof d.ui.phaseDrawerQuery === 'string' ? d.ui.phaseDrawerQuery : '';
        state.ui.fontScale = clampScale(d.ui.fontScale, 1);
        state.ui.sideOpen = d.ui.sideOpen || {};
        state.ui.sideStepOpen = d.ui.sideStepOpen || {};
        if (Array.isArray(d.ui.markFilters)) state.ui.markFilters = d.ui.markFilters.filter(x => x === 'archived' || x === 'review');
        state.ui.sideClosed = Object.assign({ dev: true }, d.ui.sideClosed || {});
      }
      nbMigrate();
      state.ui.stale = !!(d.contentVersion && d.contentVersion !== CONTENT_VERSION);
      // اگر محتوا به روز شده و کاربر ساختار را دست کاری نکرده، خودکار به روزرسانی کن
      if (state.ui.stale && !d.customized) {
        state.plan = clone(DEFAULT_PLAN);
        state.contentVersion = CONTENT_VERSION;
        state.ui.stale = false;
      }
    } catch (e) { /* داده خراب → پیش فرض */ }
  }

    state.contentVersion = CONTENT_VERSION;

  /* ---------- شمارش ها -------------------------------------------------- */
  function stepDone(s) {
    if (!s.subs.length) return !!state.checks[s.id];
    return s.subs.every(x => !!state.checks[x.id]) && !!state.checks[s.id];
  }
  function stepPartial(s) {
    return s.subs.some(x => !!state.checks[x.id]) && !stepDone(s);
  }
  function hasNote(s) {
    return !!state.notes['n:' + s.id] || s.subs.some(x => !!state.notes['n:' + x.id]);
  }
  function counts(phase) {
    let d = 0, t = 0, notes = 0;
    (phase ? [phase] : state.plan).forEach(ph => {
      ph.steps.forEach(s => {
        t++; if (state.checks[s.id]) d++;
        if (state.notes['n:' + s.id]) notes++;
        s.subs.forEach(x => {
          t++; if (state.checks[x.id]) d++;
          if (state.notes['n:' + x.id]) notes++;
        });
      });
    });
    return { d, t, notes, p: pct(d, t) };
  }

  /* ---------- به روزرسانی سبک (بدون رندر کامل) ------------------------- */
  function refreshStep(sid, opts) {
    const f = findStep(sid); if (!f) return;
    const el = document.getElementById(sid);
    if (el) {
      const done = stepDone(f.s), part = stepPartial(f.s);
      el.classList.toggle('done', done);
      const cb = el.querySelector('.st-head input[data-ck]'); if (cb) cb.checked = done;
      const tag = el.querySelector('[data-subcount]');
      if (tag) tag.textContent = fa(f.s.subs.filter(x => state.checks[x.id]).length) + ' از ' + fa(f.s.subs.length) + ' زیرمرحله';
      const pt = el.querySelector('[data-partial]'); if (pt) pt.style.display = part ? '' : 'none';
      if (!opts || opts.subs !== false) {
        f.s.subs.forEach(x => {
          const li = document.getElementById(x.id);
          if (!li) return;
          li.classList.toggle('done', !!state.checks[x.id]);
          const c = li.querySelector('input[data-ck]');
          if (c) c.checked = !!state.checks[x.id];
        });
      }
    }
    refreshPhase(f.ph.id);
  }
  function refreshPhase(phId) {
    const ph = findPhase(phId); if (!ph) return;
    const c = counts(ph);
    $$('[data-pbar="' + phId + '"]').forEach(e => e.style.width = c.p + '%');
    $$('[data-ppct="' + phId + '"]').forEach(e => e.innerHTML = fa(c.p) + '٪ <small>پیشرفت</small>');
    $$('[data-nav="' + phId + '"]').forEach(e => e.textContent = fa(c.d) + '/' + fa(c.t));
    const b = document.querySelector('[data-phbadge="' + phId + '"]');
    if (b) {
      const done = c.t > 0 && c.d === c.t;
      b.className = 'ph-badge' + (done ? ' ok' : '');
      b.textContent = done ? 'کامل شد ✓' : fa(c.d) + '/' + fa(c.t);
    }
    const sect = document.getElementById(phId);
    if (sect) {
      sect.classList.toggle('done', c.t > 0 && c.d === c.t);
      const phaseBox = sect.querySelector('.ph-ck input[data-ck]');
      if (phaseBox) {
        phaseBox.checked = c.t > 0 && c.d === c.t;
        phaseBox.indeterminate = c.d > 0 && c.d < c.t;
      }
    }
    updateStats();
  }

  /* ---------- نشان ها: آرشیو و نیازمند بررسی ------------------------------
   * state.marks = { archived: {entityId: true}, review: {entityId: true} }
   * entityId می تواند شناسه فاز، مرحله یا زیرمرحله باشد.
   * آرشیو شده ها در نمای اصلی دیده نمی شوند؛ با فیلتر «آرشیو» برمی گردند.
   * نیازمند بررسی در صفحه اصلی می ماند و فقط با نشان مشخص می شود.
   * --------------------------------------------------------------------- */
  function isArchived(id) { return !!(state.marks && state.marks.archived && state.marks.archived[id]); }
  function needsReview(id) { return !!(state.marks && state.marks.review && state.marks.review[id]); }
  function markFiltersOn() { return Array.isArray(state.ui.markFilters) && state.ui.markFilters.length > 0; }
  function archivedFilterOn() { return Array.isArray(state.ui.markFilters) && state.ui.markFilters.indexOf('archived') !== -1; }
  function entityHasFilterMark(ph, s, x) {
    if (!markFiltersOn()) return true;
    return state.ui.markFilters.some(kind => {
      const bag = state.marks && state.marks[kind];
      if (!bag) return false;
      return !!(bag[ph.id] || (s && bag[s.id]) || (x && bag[x.id]) || (s && !x && s.subs.some(z => bag[z.id])));
    });
  }
  function stepArchivedHidden(ph, s) {
    if (state.ui.editing) return false;
    if (!isArchived(ph.id) && !isArchived(s.id)) return false;
    return !archivedFilterOn();
  }
  function subArchivedHidden(ph, s, x) {
    if (state.ui.editing) return false;
    if (!isArchived(ph.id) && !isArchived(s.id) && !isArchived(x.id)) return false;
    return !archivedFilterOn();
  }
  function subShown(ph, s, x) {
    if (state.ui.editing) return true;
    if (subArchivedHidden(ph, s, x)) return false;
    if (!markFiltersOn()) return true;
    return entityHasFilterMark(ph, s, x);
  }
  function markCount(kind) {
    const bag = (state.marks && state.marks[kind]) || {};
    let n = 0;
    state.plan.forEach(ph => {
      if (bag[ph.id]) n++;
      ph.steps.forEach(s => { if (bag[s.id]) n++; s.subs.forEach(x => { if (bag[x.id]) n++; }); });
    });
    return n;
  }
  function markBadges(id) {
    let h = '';
    if (needsReview(id)) h += '<span class="tag review-tag" title="در فهرست نیازمند بررسی است">' + I('flag') + ' نیازمند بررسی</span>';
    if (isArchived(id)) h += '<span class="tag arch-tag" title="آرشیو شده است">' + I('archive') + ' آرشیو شده</span>';
    return h;
  }
  function entityBtn(etype, eid) {
    return '<button class="iact ent-btn" data-act="entity-menu" data-etype="' + etype + '" data-eid="' + esc(eid) + '" title="منوی عملیات: آرشیو، نیازمند بررسی، اشتراک و دفترچه" aria-label="باز کردن منوی عملیات">' + I('more') + '</button>';
  }
  function findEntity(etype, eid) {
    if (etype === 'phase') { const ph = findPhase(eid); return ph ? { type: 'phase', ph: ph, label: phaseNumLabel(ph) + ' — ' + stripPhasePrefix(ph.title) } : null; }
    if (etype === 'step') { const f = findStep(eid); return f ? { type: 'step', ph: f.ph, s: f.s, label: stepNumLabel(f.ph, f.ph.steps.indexOf(f.s)) + ' — ' + stripStepPrefix(f.s.title) } : null; }
    if (etype === 'sub') {
      const f = findStepOfSub(eid);
      if (!f) return null;
      const x = f.s.subs.filter(z => z.id === eid)[0];
      return x ? { type: 'sub', ph: f.ph, s: f.s, x: x, label: subNumLabel(f.ph, f.ph.steps.indexOf(f.s), f.s.subs.indexOf(x)) + ' — ' + x.title } : null;
    }
    // تشخیص خودکار بر اساس شناسه
    return findEntity('phase', eid) || findEntity('step', eid) || findEntity('sub', eid);
  }

  /* ---------- جست وجو و فیلتر ------------------------------------------- */
  function hayOf(ph, s) {
    const si = ph.steps.indexOf(s);
    const sig = ph.id + ':' + si + ':' + state.plan.indexOf(ph);
    let h = HAY.get(s);
    if (h == null || h.phId !== sig) {
      const parts = [phaseNumLabel(ph), stripPhasePrefix(ph.title), ph.goal, stepNumLabel(ph, si), stripStepPrefix(s.title), s.meta, s.tip, s.stack || ''];
      s.subs.forEach((x, i) => { parts.push(subNumLabel(ph, si, i)); parts.push(x.title); });
      h = norm(parts.join(' \u00b6 '));
      HAY.set(s, { phId: sig, t: h });
      return h;
    }
    return h.t;
  }
  function hitObj(ph, s) {
    const q = norm(state.ui.q);
    if (!q) return true;
    return hayOf(ph, s).indexOf(q) !== -1;
  }
  function hitStack(ph, s) {
    const st = norm(state.ui.stack);
    if (!st) return true;
    return hayOf(ph, s).indexOf(st) !== -1;
  }
  function visible(s) {
    const f = state.ui.filter;
    if (f === 'todo') return !stepDone(s);
    if (f === 'done') return stepDone(s);
    if (f === 'notes') return hasNote(s);
    return true;
  }
  function matchesAll(ph, s) {
    if (stepArchivedHidden(ph, s)) return false;
    if (markFiltersOn() && !entityHasFilterMark(ph, s, null)) return false;
    return visible(s) && hitObj(ph, s) && hitStack(ph, s);
  }
  function isFiltering() { return !!state.ui.q.trim() || state.ui.filter !== 'all' || !!state.ui.stack || markFiltersOn(); }
  function stackCount(st) {
    let n = 0;
    const t = norm(st);
    state.plan.forEach(ph => ph.steps.forEach(s => { if (hayOf(ph, s).indexOf(t) !== -1) n++; }));
    return n;
  }

  /* ---------- هایلایت استک فعال + شماره گذاری خودکار ----------------------
   * وقتی فیلتر استک فعال است، خودِ کلمه استک در عنوان/نکته/زیرمرحله ها با
   * <mark class="hl-stack"> برجسته می شود تا کاربر دلیل نمایش هر مرحله را ببیند.
   * شماره فاز/مرحله/زیرمرحله «محاسباتی» است (از روی جایگاه فعلی در آرایه)؛
   * پس با افزودن، حذف یا جابه جایی، همه شماره ها خودکار اصلاح می شوند و
   * هیچ شماره ای داخل متن عنوان به صورت دستی ذخیره نمی شود.
   * --------------------------------------------------------------------- */
  function hiStack(text) {
    const raw = String(text == null ? '' : text);
    const term = String(state.ui.stack || '').trim();
    if (!term || state.ui.editing) return esc(raw);
    const low = raw.toLowerCase(), tl = term.toLowerCase();
    let out = '', i = 0, j = low.indexOf(tl);
    if (j === -1) return esc(raw);
    while (j !== -1) {
      out += esc(raw.slice(i, j)) + '<mark class="hl-stack">' + esc(raw.slice(j, j + term.length)) + '</mark>';
      i = j + term.length; j = low.indexOf(tl, i);
    }
    return out + esc(raw.slice(i));
  }
  function isAppendixPhase(ph) {
    return !!ph && (ph.id === 'ap' || /^\s*پیوست/.test(String(ph.title || '')));
  }
  function phaseOrdinal(ph) {
    let n = 0;
    for (const p of state.plan) {
      if (isAppendixPhase(p)) continue;
      n++;
      if (p.id === ph.id) return n;
    }
    return null;
  }
  function phaseNumLabel(ph) {
    if (isAppendixPhase(ph)) return 'پیوست';
    const n = phaseOrdinal(ph);
    return 'فاز ' + fa(n == null ? 0 : n);
  }
  function stripPhasePrefix(title) {
    return String(title || '').replace(/^\s*فاز\s*[۰-۹0-9]+\s*[—\-–:]\s*/, '').replace(/^\s*پیوست ها\s*[—\-–:]\s*/, '');
  }
  function stripStepPrefix(title) {
    return String(title || '').replace(/^\s*(پیوست|مرحله|گام)\s*[۰-۹0-9]+\s*[—\-–:]\s*/, '');
  }
  function stepNumLabel(ph, si) {
    if (isAppendixPhase(ph)) return fa(si + 1);
    const n = phaseOrdinal(ph);
    return fa(n == null ? 0 : n) + '.' + fa(si + 1);
  }
  function subNumLabel(ph, si, i) {
    if (isAppendixPhase(ph)) return fa(i + 1);
    const n = phaseOrdinal(ph);
    return fa(n == null ? 0 : n) + '.' + fa(si + 1) + '.' + fa(i + 1);
  }


  /* ---------- منوی عملیات هر مورد و ناوبری مطمئن -------------------------
   * برای شلوغ نشدن صفحه، هر فاز و مرحله و زیرمرحله فقط یک دکمه کوچک «...»
   * دارد و چهار عمل در یک منوی شناور باز می شود:
   * آرشیو | نیازمند بررسی | اشتراک با URL | اضافه کردن به دفترچه
   * --------------------------------------------------------------------- */
  let entCtx = null;
  let entityMenuEl = null;
  function entMenuMount() {
    if (entityMenuEl) return;
    const d = document.createElement('div');
    d.id = 'entityMenu';
    d.className = 'entity-menu';
    d.hidden = true;
    document.body.appendChild(d);
    entityMenuEl = d;
    d.addEventListener('click', (e) => {
      const b = e.target.closest('[data-entact]'); if (!b || !entCtx) return;
      const act = b.dataset.entact, ctx = entCtx;
      closeEntityMenu();
      if (act === 'archive') toggleEntityMark('archived', ctx);
      else if (act === 'review') toggleEntityMark('review', ctx);
      else if (act === 'share') shareEntity(ctx);
      else if (act === 'notebook') addEntityToNotebook(ctx);
    });
    document.addEventListener('click', (e) => {
      if (entityMenuEl && !entityMenuEl.hidden && !e.target.closest('#entityMenu') && !e.target.closest('[data-act="entity-menu"]')) closeEntityMenu();
    });
    document.addEventListener('keydown', (e) => { if (e.key === 'Escape') closeEntityMenu(); });
    window.addEventListener('resize', closeEntityMenu);
  }
  function closeEntityMenu() { if (entityMenuEl) entityMenuEl.hidden = true; entCtx = null; }
  function openEntityMenu(btn, etype, eid) {
    entMenuMount();
    const ent = findEntity(etype, eid);
    if (!ent) { toast('این مورد پیدا نشد.'); return; }
    entCtx = { type: etype, id: eid };

    const r = btn.getBoundingClientRect();

    if (left + w > window.innerWidth - 8) left = window.innerWidth - w - 8;

    if (top + 250 > window.innerHeight) top = Math.max(8, r.top - 256);

  function toggleEntityMark(kind, ctx) {
    if (!state.marks) state.marks = { archived: {}, review: {} };
    if (!state.marks[kind]) state.marks[kind] = {};
    const on = !!state.marks[kind][ctx.id];
    if (on) delete state.marks[kind][ctx.id]; else state.marks[kind][ctx.id] = true;
    saveNow(); render();
    if (kind === 'archived') toast(on ? 'از آرشیو خارج شد و در صفحه اصلی دیده می شود.' : 'آرشیو شد؛ با فیلتر «آرشیو» در هدر یا سایدبار دوباره دیده می شود.');
    else toast(on ? 'از فهرست نیازمند بررسی حذف شد.' : 'به فهرست نیازمند بررسی اضافه شد.');
  }
  function entityShareUrl(ctx) {
    const base = String(location.href || '').split('#')[0];
    return base + '#item=' + encodeURIComponent(ctx.type + ':' + ctx.id);
  }
  function shareEntity(ctx) {
    const ent = findEntity(ctx.type, ctx.id); if (!ent) return;
    const url = entityShareUrl(ctx);
    try { history.replaceState(null, '', '#item=' + encodeURIComponent(ctx.type + ':' + ctx.id)); } catch (e) { }
    const title = ent.label;
    const done = () => toast('لینک اشتراک کپی شد: ' + url);
    const copyIt = () => {
      if (navigator.clipboard && navigator.clipboard.writeText) navigator.clipboard.writeText(url).then(done, () => fallbackCopy(url, done));
      else fallbackCopy(url, done);
    };
    try {
      if (navigator.share) {
        navigator.share({ title: title, text: title, url: url }).then(() => toast('اشتراک گذاری انجام شد.'), (e) => { if (!e || e.name !== 'AbortError') copyIt(); });
        return;
      }
    } catch (e) { }
    copyIt();
  }
  function addEntityToNotebook(ctx) {
    const ent = findEntity(ctx.type, ctx.id); if (!ent) return;
    nbMigrate();
    const lines = [];
    let title = ent.label;
    if (ent.type === 'phase') {
      lines.push('# ' + ent.label, '');
      if (ent.ph.goal) lines.push('> ' + ent.ph.goal, '');
      if (ent.ph.dod) lines.push('**شرط اتمام فاز:** ' + ent.ph.dod, '');
      ent.ph.steps.forEach((s, si) => {
        lines.push('## ' + stepNumLabel(ent.ph, si) + ' — ' + stripStepPrefix(s.title));
        s.subs.forEach((x, i) => lines.push('- [' + (state.checks[x.id] ? 'x' : ' ') + '] ' + subNumLabel(ent.ph, si, i) + ' ' + x.title));
        lines.push('');
      });
    } else if (ent.type === 'step') {
      const s = ent.s, si = ent.ph.steps.indexOf(s);
      lines.push('# ' + ent.label, '', '**فاز:** ' + phaseNumLabel(ent.ph) + ' — ' + stripPhasePrefix(ent.ph.title));
      if (s.meta) lines.push('**مشخصات:** ' + s.meta);
      if (s.stack) lines.push('**استک:** ' + s.stack);
      if (s.tip) lines.push('', '> نکته: ' + s.tip);
      lines.push('', '## زیرمرحله ها');
      s.subs.forEach((x, i) => lines.push('- [' + (state.checks[x.id] ? 'x' : ' ') + '] ' + subNumLabel(ent.ph, si, i) + ' ' + x.title));
      const n = state.notes['n:' + s.id];
      if (n) lines.push('', '## یادداشت فعلی مرحله', n);
    } else {
      const s = ent.s, x = ent.x, si = ent.ph.steps.indexOf(s);
      lines.push('# ' + ent.label, '', '**فاز:** ' + phaseNumLabel(ent.ph) + ' — ' + stripPhasePrefix(ent.ph.title));
      lines.push('**مرحله:** ' + stepNumLabel(ent.ph, si) + ' — ' + stripStepPrefix(s.title));
      lines.push('**وضعیت:** ' + (state.checks[x.id] ? 'انجام شده' : 'باقی مانده'));
      const n = state.notes['n:' + x.id];
      if (n) lines.push('', '## یادداشت فعلی زیرمرحله', n);
    }
    lines.push('', '---', 'لینک این مورد: ' + entityShareUrl(ctx));
    const now = new Date().toISOString();
    const note = { id: uid('nb'), title: title.slice(0, 80), body: lines.join('\n'), createdAt: now, updatedAt: now };
    state.nbNotes.push(note);
    state.nbActiveId = note.id;

  /* ناوبری مطمئن سایدبار و لینک مستقیم: حتی بعد از چند جستجو و فیلتر،
   * اگر مورد با فیلتر فعلی پنهان باشد فیلتر ها پاک می شوند و مورد باز می شود. */
  function scrollToId(id) {
    const doScroll = () => {
      const el = document.getElementById(id);
      if (!el) return false;
      const ph = el.closest ? el.closest('.phase') : null;
      if (ph) ph.classList.add('open');
      const r = el.getBoundingClientRect();
      const offset = el.classList.contains('phase') ? 106 : (el.classList.contains('step') ? 128 : 150);
      const target = Math.max(0, window.scrollY + r.top - offset);
      // jump is deliberate for long pages: a very long smooth animation looked like a failed nav
      const root = document.documentElement, old = root.style.scrollBehavior;
      root.style.scrollBehavior = 'auto';
      try { window.scrollTo(0, target); } catch (e) { try { el.scrollIntoView({ behavior: 'instant', block: 'start' }); } catch (_) { if (el.scrollIntoView) el.scrollIntoView(); } }
      root.style.scrollBehavior = old;
      el.classList.add('hit');
      setTimeout(() => el.classList.remove('hit'), 1800);
      return true;
    };
    if (!doScroll()) { setTimeout(doScroll, 80); setTimeout(doScroll, 260); }
    else { requestAnimationFrame(() => requestAnimationFrame(doScroll)); }
  }
  function navigateToEntity(etype, eid) {
    const ent = findEntity(etype, eid);
    if (!ent) { toast('این مورد در نقشه راه پیدا نشد.'); return false; }
    const targetId = eid;
    const arch = isArchived(ent.ph.id) || (ent.s && isArchived(ent.s.id)) || (ent.x && isArchived(ent.x.id));
    // پاک کردن فیلتر هایی که مورد را پنهان می کنند
    state.ui.q = '';
    state.ui.filter = 'all';
    state.ui.stack = '';
    state.ui.markFilters = arch ? ['archived'] : [];
    state.open[ent.ph.id] = true;
    if (!state.ui.sideOpen) state.ui.sideOpen = {};
    state.ui.sideOpen[ent.ph.id] = true;
    save(); render();
    scrollToId(targetId);
    return true;
  }
  function handleDeepLink() {
    const raw = String(location.hash || '').replace(/^#/, '');
    if (!raw) return;
    let m = null;
    try { m = decodeURIComponent(raw).match(/item=(phase|step|sub):(.+)$/); } catch (e) { }
    if (m) { navigateToEntity(m[1], m[2]); return; }
    if (document.getElementById(raw)) navigateToEntity('auto', raw);
  }

  function renderInlineText(text, allowBlockCode) {
    const value = String(text == null ? '' : text);
    if (state.ui.editing) return esc(value).replace(/\n/g, '<br>');
    if (allowBlockCode && /(?:^|\n)\s*(```|~~~)/.test(value)) return mdToHtml(value);
    return mdInline(value);

  function refreshFontScales() {
    document.documentElement.style.setProperty('--app-font-scale', String(clampScale(state.ui.fontScale, 1)));
    document.documentElement.style.setProperty('--nb-font-scale', String(clampScale(state.nbFontScale, 1)));

  function adjustFontScale(scope, dir) {
    const isNb = scope === 'nb';
    const key = isNb ? 'nbFontScale' : null;
    const current = isNb ? state.nbFontScale : state.ui.fontScale;
    const next = Number(dir) === 0 ? 1 : clampScale(current + (Number(dir) < 0 ? -.05 : .05), 1);

    refreshFontScales(); saveNow();
  }
  function renderPhaseTree(navId, query) {
    const q = norm(query || '');

      const pText = norm([phaseNumLabel(ph), ph.title, ph.goal, ph.dod].join(' '));
      const pHit = !q || pText.indexOf(q) !== -1;
      const stepMatches = ph.steps.map((s, si) => {

        const matchingSubs = s.subs.map((x, xi) => ({ x, xi })).filter(o => norm([subNumLabel(ph, si, o.xi), o.x.title].join(' ')).indexOf(q) !== -1);
        return { s, si, hit: !q || pHit || text.indexOf(q) !== -1 || matchingSubs.length > 0, textHit: !q || pHit || text.indexOf(q) !== -1, matchingSubs };

  function expandDrawerSearch(q) {
    const query = norm(q);
    if (!query) return;
    state.plan.forEach(ph => {
      const pHit = norm([phaseNumLabel(ph), ph.title, ph.goal, ph.dod].join(' ')).includes(query);
      const matching = [];
      ph.steps.forEach((s, si) => {
        const sHit = norm([stepNumLabel(ph, si), s.title, s.meta, s.tip, s.stack].join(' ')).includes(query);
        const subHits = s.subs.some((x, xi) => norm([subNumLabel(ph, si, xi), x.title].join(' ')).includes(query));
        if (sHit || subHits) matching.push({s,subHits});
      });
      if (pHit || matching.length) state.ui.sideOpen[ph.id] = true;
      matching.forEach(o => { if (o.subHits) state.ui.sideStepOpen[o.s.id] = true; });
    });
  }

  /* درج در جایگاه دلخواه؛ مقدار خالی رفتار پیش فرض را حفظ می کند. */
  function parseInsertPosition(raw, label, count) {
    if (raw == null || !String(raw).trim()) return { ok:true, index:null };
    const m = norm(raw).match(/^([0-9]+)$/);
    if (!m) { toast('در «محل درج ' + label + '» فقط عدد وارد کن.'); return {ok:false,index:null}; }
    const n = Number(m[1]);
    if (!Number.isInteger(n) || n < 1 || n > count + 1) {
      toast('محل درج ' + label + ' باید بین ۱ تا ' + fa(count + 1) + ' باشد.');
      return {ok:false,index:null};
    }
    return {ok:true,index:n - 1};
  }
  function addStep(phId, afterStepId, position) {
    const ph = findPhase(phId); if (!ph) return;
    const pos = parseInsertPosition(position, 'مرحله', ph.steps.length);
    if (!pos.ok) return;
    const s = { id: uid('st'), title: 'مرحله جدید — عنوان را در حالت ویرایش تغییر بده', meta: '≈ ؟ · مسئول', tip: '', subs: [] };
    if (pos.index != null) ph.steps.splice(pos.index, 0, s);
    else if (afterStepId && typeof afterStepId === 'string') {
      const i = ph.steps.map(x => x.id).indexOf(afterStepId);
      ph.steps.splice(i + 1, 0, s);
    } else ph.steps.push(s);
    state.open[phId] = true; touch(); render();
    flash(s.id);
    toast('مرحله جدید در جایگاه ' + fa(ph.steps.indexOf(s) + 1) + ' درج شد؛ شماره بقیه مراحل خودکار به روز شد.');
  }
  function addSub(stepId, after, position) {
    const f = findStep(stepId); if (!f) return;
    const list = f.s.subs;
    const pos = parseInsertPosition(position, 'زیرمرحله', list.length);
    if (!pos.ok) return;
    const item = { id: uid('sb'), title: 'زیرمرحله جدید — متن را ویرایش کن' };
    if (pos.index != null) list.splice(pos.index, 0, item);
    else {
      const i = Number(after);
      if (i >= 0 && i < list.length) list.splice(i + 1, 0, item); else list.push(item);
    }
    touch(); render(); flash(item.id);
    toast('زیرمرحله جدید در جایگاه ' + fa(list.indexOf(item) + 1) + ' درج شد؛ شماره بقیه خودکار به روز شد.');
  }

  const BOM = '\uFEFF';   // برای نمایش درست فارسی در Notepad ویندوز
  function download(name, text, type) {
    try {
      const isText = /json|markdown|text/i.test(type || 'application/json');
      const body = (isText && /markdown|text\/plain/i.test(type || '')) ? (BOM + text) : text;
      const b = new Blob([body], { type: (type || 'application/json') + ';charset=utf-8' });
      const url = URL.createObjectURL(b);
      const a = document.createElement('a');
      a.href = url; a.download = name; a.rel = 'noopener';
      document.body.appendChild(a); a.click(); a.remove();
      setTimeout(() => URL.revokeObjectURL(url), 8000);
      toast('دانلود شروع شد: ' + name + ' (اگر مسدود شد، از پنجره «متن گزارش» کپی کن)');
      return true;
    } catch (e) {
      toast('مرورگر اجازه دانلود نداد — متن در پنجره «متن گزارش» آماده است.');
      return false;
    }
  }
  /* باز کردن متن در تب جدید (اگر مرورگر اجازه دهد) — جایگزین «باز نشدن فایل» */
  function openInTab(text, type) {
    try {
      const b = new Blob([BOM + text], { type: (type || 'text/plain') + ';charset=utf-8' });
      const url = URL.createObjectURL(b);
      const w = window.open(url, '_blank');
      if (!w) { toast('باز کردن تب جدید مسدود شد (پیش نمایش داخلی). از «کپی متن» استفاده کن.'); return false; }
      setTimeout(() => URL.revokeObjectURL(url), 30000);
      return true;
    } catch (e) { toast('باز کردن تب جدید ممکن نشد — از «کپی متن» استفاده کن.'); return false; }
  }

        } else if (mode === 'todo') {

          stepTag = '[ ]';
        } else {

          stepTag = isDone ? '[x]' : (doneSubs.length ? '[~]' : '[ ]');
        }

          if (sn && mode !== 'todo') buf.push('    - 🗒 یادداشت: ' + String(sn).replace(/\n/g, ' '));
        });

        if (stn) buf.push('\n> 📝 **یادداشت مرحله:** ' + String(stn).replace(/\n/g, '\n> '));
      });
      if (buf.length) {

        if (mode === 'full' && ph.goal) md += '> ' + ph.goal + '\n';
        if (mode === 'full' && ph.dod) md += '\n**شرط اتمام فاز:** ' + ph.dod + '\n';
        md += buf.join('\n') + '\n';
      }
    });
    if (!any) md += '\n_(موردی برای این حالت گزارش وجود ندارد.)_\n';
    const nbList = nbSorted((state.nbNotes || []).filter(n => String(n.body || '').trim() || String(n.title || '').trim()));
    if (nbList.length) {
      md += '\n---\n\n## 📝 دفترچه یادداشت (مستقل) — ' + fa(nbList.length) + ' یادداشت\n';
      nbList.forEach((n, i) => {
        md += '\n### ' + fa(i + 1) + '. ' + (String(n.title || '').trim() || nbTitleFromBody(n.body)) + '\n\n' + String(n.body || '').trim() + '\n';
      });
    } else if ((state.notebook || '').trim()) {
      md += '\n---\n\n## 📝 دفترچه یادداشت (مستقل)\n\n' + state.notebook.trim() + '\n';
    }
    md += '\n---\n\n_ساخته شده با نقشه راه ' + BRAND + ' — ' + faDate + '_\n';
    return md;
  }

  /* --- رندر مینیمال Markdown (بدون وابستگی خارجی) --- */
  function highlightStackEscaped(text) {
    const term = String(state.ui.stack || '').trim();
    if (!term) return text;
    const needle = esc(term).replace(/[.*+?^${}()|[\]\\]/g, '\\$&');
    if (!needle) return text;
    try { return text.replace(new RegExp(needle, 'gi'), m => '<mark class="hl-stack">' + m + '</mark>'); }
    catch (e) { return text; }
  }
  function mdInline(t) {
    let x = highlightStackEscaped(esc(t));
    x = x.replace(/`([^`]+)`/g, '<code>$1</code>');
    x = x.replace(/\*\*([^*]+)\*\*/g, '<strong>$1</strong>');
    x = x.replace(/(^|[^*])\*([^*\n]+)\*/g, '$1<em>$2</em>');
    x = x.replace(/~~([^~]+)~~/g, '<del>$1</del>');
    x = x.replace(/==([^=]+)==/g, '<mark>$1</mark>');
    x = x.replace(/\[([^\]]+)\]\(([^)\s]+)\)/g, function (m, txt, href) {
      const h = href.replace(/&amp;/g, '&');
      if (!/^(https?:|mailto:|#|\/)/i.test(h)) return m;
      return '<a href="' + h + '" target="_blank" rel="noopener noreferrer">' + txt + '</a>';
    });
    x = x.replace(/(^|[\s(])(https?:\/\/[^\s<)]+)/g, function (m, p, u) {
      return p + '<a href="' + u + '" target="_blank" rel="noopener noreferrer">' + u + '</a>';
    });
    return x;
  }
  function highlightCode(source, language) {
    const src = String(source || ''), lang0 = String(language || '').toLowerCase();
    const aliases = { js:'javascript', jsx:'javascript', mjs:'javascript', cjs:'javascript', ts:'typescript', tsx:'typescript', py:'python', rb:'ruby', sh:'bash', zsh:'bash', yml:'yaml', xml:'html', xhtml:'html', htm:'html', h:'c', cxx:'cpp', 'c++':'cpp' };
    const lang = aliases[lang0] || lang0;
    const keywords = {
      javascript:'const let var function return if else for while do switch case break continue try catch finally throw new class extends import export from as await async yield this super typeof instanceof in of true false null undefined delete void debugger',
      typescript:'const let var function return if else for while do switch case break continue try catch finally throw new class extends import export from as await async yield this super typeof instanceof in of true false null undefined type interface enum implements public private protected readonly namespace module declare keyof unknown never any string number boolean',
      python:'def class return if elif else for while break continue try except finally raise with as import from lambda yield async await self True False None and or not in is pass global nonlocal assert del match case',
      php:'function return if else elseif foreach for while do switch case break continue try catch finally throw new class extends implements public private protected static use namespace echo print true false null as fn',
      bash:'if then else elif fi for while do done case esac function return local export readonly in echo cd pwd printf read set unset exit source',
      sql:'select from where and or not null is like in join left right inner outer on as group by order having limit offset insert into values update set delete create alter table primary key foreign references drop distinct count sum avg min max',
      css:'important inherit initial unset auto none block flex grid inline position absolute relative fixed sticky color background display width height margin padding border font opacity overflow transform',
      json:'true false null',
      html:'doctype html head body div span section article main aside header footer nav button input form label script style link meta title class id type name value href src role aria-label data-id'
    };
    const types = new Set('Array Object String Number Boolean Date Map Set Promise Error RegExp Math JSON HTMLElement Element Event NodeList'.split(' '));
    const wordSet = new Set((keywords[lang] || keywords.javascript).split(/\s+/));
    const lineHash = ['python','bash','yaml','ruby','php'].includes(lang);
    const lineDash = lang === 'sql';
    const isJs = ['javascript','typescript','java','c','cpp','csharp','go','rust','swift','kotlin'].includes(lang);
    let i = 0, out = '';
    const span = (cls, text) => '<span class="hl-' + cls + '">' + esc(text) + '</span>';
    const starts = (x) => src.slice(i, i + x.length) === x;
    while (i < src.length) {
      const ch = src[i];
      if (starts('/*')) {
        const end = src.indexOf('*/', i + 2), j = end < 0 ? src.length : end + 2;
        out += span('comment', src.slice(i,j)); i=j; continue;
      }
      if (lang === 'html' && starts('<!--')) {
        const end = src.indexOf('-->', i + 4), j = end < 0 ? src.length : end + 3;
        out += span('comment', src.slice(i,j)); i=j; continue;
      }
      if (starts('//') || (lineDash && starts('--')) || (lineHash && ch === '#')) {
        let j = src.indexOf('\n', i); if (j < 0) j = src.length;
        out += span('comment', src.slice(i,j)); i=j; continue;
      }
      if (ch === '"' || ch === "'" || ch === '`') {
        const quote = ch; let j = i + 1;
        while (j < src.length) { if (src[j] === '\\') { j += 2; continue; } if (src[j] === quote) { j++; break; } j++; }
        const token = src.slice(i,j);
        const after = src.slice(j).match(/^\s*:/);
        const cls = lang === 'json' && after ? 'property' : 'string';
        out += span(cls, token); i=j; continue;
      }
      const number = src.slice(i).match(/^(?:0x[0-9a-f]+|(?:\d+\.?\d*|\.\d+)(?:e[+-]?\d+)?)/i);
      if (number) { out += span('number', number[0]); i += number[0].length; continue; }
      const word = src.slice(i).match(/^[A-Za-z_$][\w$-]*/);
      if (word) {
        const token = word[0], after = src.slice(i + token.length).match(/^\s*([(:])/), prev = src.slice(0,i).match(/(?:<\/?|\s)([\w-]+)$/);
        let cls = '';
        if (lang === 'json' && src.slice(i + token.length).match(/^\s*:/)) cls = 'property';
        else if (wordSet.has(token) || wordSet.has(token.toLowerCase())) cls = 'keyword';
        else if (types.has(token) || (/^[A-Z]/.test(token) && ['python','php','typescript','javascript'].includes(lang))) cls = 'type';
        else if (lang === 'css' && src.slice(i + token.length).match(/^\s*:/)) cls = 'property';
        else if (lang === 'html' && /<\/?$/.test(src.slice(0,i))) cls = 'tag';
        else if (lang === 'html' && src.slice(i + token.length).match(/^\s*=/)) cls = 'property';
        else if (after && after[1] === '(') cls = 'function';
        else if (isJs && /^[A-Z]/.test(token)) cls = 'type';
        out += cls ? span(cls,token) : esc(token); i += token.length; continue;
      }
      if (/[=+*/%<>!&|^~?:-]/.test(ch)) {
        const op = src.slice(i).match(/^(?:===|!==|=>|==|!=|<=|>=|&&|\|\||\?\?|\+\+|--|\+=|-=|\*=|\/=|\*\*|->|::|[=+*/%<>!&|^~?:-])/)[0];
        out += span('operator',op); i += op.length; continue;
      }
      out += esc(ch); i++;
    }
    return out;
  }
  function mdToHtml(src) {
    const lines = String(src || '').replace(/\r\n?/g, '\n').split('\n');
    const out = []; let i = 0;
    const cellSplit = (r) => r.trim().replace(/^\||\|$/g, '').split('|').map(c => c.trim());
    while (i < lines.length) {
      const ln = lines[i];
      const fence = ln.match(/^\s*(```|~~~)\s*([a-z0-9_+#.-]*)\s*$/i);
      if (fence) {
        const marker = fence[1], lang = (fence[2] || '').toLowerCase(); i++; const buf = [];
        while (i < lines.length && !new RegExp('^\\s*' + marker.replace(/[.*+?^${}()|[\]\\]/g, '\\$&')).test(lines[i])) { buf.push(lines[i]); i++; }
        if (i < lines.length) i++;
        const langAttr = lang || 'code';
        out.push('<pre class="md-pre" data-lang="' + esc(langAttr) + '"><code class="language-' + esc(langAttr) + '">' + highlightCode(buf.join('\n'),lang) + '</code></pre>'); continue;
      }
      if (/^\s*$/.test(ln)) { i++; continue; }
      if (/^\s*(-{3,}|\*{3,}|_{3,})\s*$/.test(ln)) { out.push('<hr>'); i++; continue; }
      let m = ln.match(/^(#{1,6})\s+(.*)$/);
      if (m) { const lv = m[1].length; out.push('<h' + lv + '>' + mdInline(m[2]) + '</h' + lv + '>'); i++; continue; }
      if (/^\s*>/.test(ln)) {
        const buf = [];
        while (i < lines.length && /^\s*>/.test(lines[i])) { buf.push(lines[i].replace(/^\s*>\s?/, '')); i++; }
        out.push('<blockquote>' + mdInline(buf.join('\n')).replace(/\n/g, '<br>') + '</blockquote>'); continue;
      }
      if (/^\s*\|.*\|\s*$/.test(ln) && i + 1 < lines.length && /^\s*\|[\s:\-|]+\|\s*$/.test(lines[i + 1])) {
        const head = cellSplit(ln); i += 2; const rows = [];
        while (i < lines.length && /^\s*\|.*\|\s*$/.test(lines[i])) { rows.push(cellSplit(lines[i])); i++; }
        out.push('<table class="md-table"><thead><tr>' + head.map(h => '<th>' + mdInline(h) + '</th>').join('') + '</tr></thead><tbody>' +
          rows.map(r => '<tr>' + r.map(c => '<td>' + mdInline(c) + '</td>').join('') + '</tr>').join('') + '</tbody></table>');
        continue;
      }
      if (/^\s*([-*+]|\d+[.)])\s+/.test(ln)) {
        const ordered = /^\s*\d+[.)]\s+/.test(ln);
        const items = [];
        while (i < lines.length && /^\s*([-*+]|\d+[.)])\s+/.test(lines[i])) {
          items.push(lines[i].replace(/^\s*([-*+]|\d+[.)])\s+/, '')); i++;
        }
        out.push('<' + (ordered ? 'ol' : 'ul') + '>' + items.map(t => {
          const tm = t.match(/^\[( |x|X)\]\s*(.*)$/);
          if (tm) return '<li class="md-task">' + (tm[1].toLowerCase() === 'x' ? '☑' : '☐') + ' ' + mdInline(tm[2]) + '</li>';
          return '<li>' + mdInline(t) + '</li>';
        }).join('') + '</' + (ordered ? 'ol' : 'ul') + '>');
        continue;
      }
      const buf = [ln]; i++;
      while (i < lines.length && !/^\s*$/.test(lines[i]) &&
             !/^\s*(#{1,6}\s|>|[-*+]\s|\d+[.)]\s|```|\|)/.test(lines[i])) { buf.push(lines[i]); i++; }
      out.push('<p>' + mdInline(buf.join('\n')).replace(/\n/g, '<br>') + '</p>');
    }
    return out.join('\n') || '<p class="md-empty">هنوز چیزی نوشته نشده…</p>';
  }

  function nbFormat(kind, side) {
    side = side === 'secondary' ? 'secondary' : 'primary';
    if (nbMode(side) === 'read') nbToggleMode('edit', side);
    const ta = side === 'secondary' ? nbEls.editSecondary : nbEls.edit; if (!ta) return;
    const s0 = ta.selectionStart, s1 = ta.selectionEnd, val = ta.value, sel = val.slice(s0, s1);
    const wrapIn = (pre, post, ph) => { const txt = sel || ph; ta.value = val.slice(0, s0) + pre + txt + post + val.slice(s1); ta.selectionStart = s0 + pre.length; ta.selectionEnd = s0 + pre.length + txt.length; };
    const linePrefix = (p) => { const ls = val.lastIndexOf('\n', s0 - 1) + 1; ta.value = val.slice(0, ls) + p + val.slice(ls); ta.selectionStart = ta.selectionEnd = s0 + p.length; };
    if (kind === 'b') wrapIn('**', '**', 'متن ضخیم');
    else if (kind === 'i') wrapIn('*', '*', 'متن کج');
    else if (kind === 'strike') wrapIn('~~', '~~', 'متن خط خورده');
    else if (kind === 'h1') linePrefix('# '); else if (kind === 'h2' || kind === 'h') linePrefix('## '); else if (kind === 'h3') linePrefix('### ');
    else if (kind === 'ul') linePrefix('- '); else if (kind === 'ol') linePrefix('1. '); else if (kind === 'cb') linePrefix('- [ ] ');
    else if (kind === 'quote') linePrefix('> '); else if (kind === 'code') wrapIn('`', '`', 'کد'); else if (kind === 'link') wrapIn('[', '](https://)', 'متن لینک');
    else if (kind === 'hr') ta.value = val.slice(0, s0) + '\n\n---\n\n' + val.slice(s1);
    else if (kind === 'table') ta.value = val.slice(0, s0) + '\n| ستون ۱ | ستون ۲ |\n|---|---|\n| مقدار | مقدار |\n' + val.slice(s1);
    const n = nbActive(side); if (n) { n.body = ta.value; n.updatedAt = new Date().toISOString(); }
    nbSyncLegacy(); ta.focus(); save(); nbRenderPane(side);
  }

  async function nbShare(side) {
    side = side === 'secondary' ? 'secondary' : 'primary';
    const rendered = state.nbView === 'note' && nbMode(side) === 'read', t = nbCurrentText(rendered, side), n = nbActive(side);
    if (!t.trim()) { toast('این یادداشت خالی است.'); return; }
    const title = (n && String(n.title || '').trim()) || nbTitleFromBody(n ? n.body : '') || ('دفترچه یادداشت ' + BRAND);
    try {
      if (!rendered && navigator.canShare && window.File) {
        const file = new File([t], 'packpolymer-note.md', { type: 'text/markdown' });
        if (navigator.canShare({ files: [file] })) { await navigator.share({ title, files: [file] }); toast('اشتراک گذاری انجام شد ✅'); return; }
      }
      if (navigator.share) { await navigator.share({ title, text: t }); toast('اشتراک گذاری انجام شد ✅'); return; }
    } catch (e) { if (e && e.name === 'AbortError') return; }
    nbCopy(side); toast('مرورگر اشتراک گذاری مستقیم ندارد — متن کپی شد؛ هر جا می خواهی بچسبان.');
  }
  function nbDownload(side) {

    const name = 'packpolymer-note-' + stamp() + '.md';
    const text = '# ' + (String(n.title || '').trim() || nbTitleFromBody(n.body)) + '\n\n' + t;
    download(name, text, 'text/markdown'); showTextModal({ title: 'یادداشت: ' + safe, text, filename: name, mode: 'notebook' });
  }
  function nbClear() { nbDeleteNote(state.nbActiveId, true, 'primary'); }


  /* ---------- commit متن های ویرایش شده --------------------------------- */
  function commitField(el) {
    const f = el.dataset.field, id = el.dataset.id;
    const val = (el.innerText || '').replace(/\u00a0/g, ' ').trim();
    if (!f || !id) return;
    if (f === 'mtitle') { state.meta.title = val || DEF_TITLE; }
    else if (f === 'msub') { state.meta.sub = val || DEF_SUB; }
    else if (f === 'sub') {
      const fs = findStep(id); if (!fs) return;
      const x = fs.s.subs[Number(el.dataset.sub)]; if (!x) return;
      x.title = val;
    } else {
      const ph = findPhase(id);
      if (ph) {
        if (f === 'icon') ph.icon = val || ph.icon;
        else if (f === 'ptitle') ph.title = val || ph.title;
        else if (f === 'goal') ph.goal = val;
        else if (f === 'dod') ph.dod = val;
        state.plan.forEach(x => x.steps.forEach(st2 => invalidateHay(st2)));
      } else {
        const fs = findStep(id);
        if (!fs) return;
        if (f === 'title') fs.s.title = val || fs.s.title;
        else if (f === 'tip') fs.s.tip = val;
        else if (f === 'meta') fs.s.meta = val;
        invalidateHay(fs.s);
      }
    }
    markCustomized(); save();
  }
  function markCustomized() { try { state.customized = true; } catch (e) { } }

  /* ---------- رویدادها -------------------------------------------------- */
  function bind() {
    const app = $('#app');

    app.addEventListener('change', (e) => {
      const ck = e.target.closest('input[data-ck]');
      if (!ck) return;
      const k = ck.dataset.ck;
      const phase = findPhase(k);
      if (phase) {
        phase.steps.forEach(s => {
          if (ck.checked) state.checks[s.id] = true; else delete state.checks[s.id];
          s.subs.forEach(x => { if (ck.checked) state.checks[x.id] = true; else delete state.checks[x.id]; });
        });
        phase.steps.forEach(s => refreshStep(s.id));
        const phaseBox = document.querySelector('.phase[data-phase="' + k + '"] .ph-ck input[data-ck]');
        if (phaseBox) phaseBox.checked = phase.steps.length > 0 && phase.steps.every(s => stepDone(s));
        save(); updateStats();
        return;
      }
      if (ck.checked) state.checks[k] = true; else delete state.checks[k];

      // منطق والد/فرزند: تیک مرحله → همه زیرمرحله ها | تیک همه زیرمرحله ها → تیک مرحله
      const fs = findStep(k);
      if (fs) {
        fs.s.subs.forEach(x => { if (ck.checked) state.checks[x.id] = true; else delete state.checks[x.id]; });
        refreshStep(k);
      } else {
        const owner = findStepOfSub(k);
        if (owner) {
          const allDone = owner.s.subs.every(x => !!state.checks[x.id]);
          if (allDone) state.checks[owner.s.id] = true; else delete state.checks[owner.s.id];
          const li = document.getElementById(k);
          if (li) { li.classList.toggle('done', !!state.checks[k]); }
          refreshStep(owner.s.id, { subs: false });
        } else {
          const subEl = document.getElementById(k);
          if (subEl) subEl.classList.toggle('done', !!state.checks[k]);
          updateStats();
        }
      }
      save(); updateStats();
    });

    app.addEventListener('input', (e) => {
      const n = e.target.closest('textarea[data-note]');
      if (n) {
        const k = n.dataset.note;
        if (n.value.trim()) state.notes[k] = n.value; else delete state.notes[k];
        n.classList.toggle('has', !!n.value.trim());
        const cnt = n.parentElement.querySelector('.cnt');
        if (cnt) cnt.textContent = n.value.trim() ? 'ذخیره شد ✍️' : 'خالی';
        save(); updateStats();
        return;
      }
      if (e.target.dataset && e.target.dataset.meta) {
        const f = findStep(e.target.dataset.meta);
        if (f) { f.s.meta = e.target.value; markCustomized(); save(); }
        return;
      }
      if (e.target.id === 'phaseDrawerSearch') {
        state.ui.phaseDrawerQuery = e.target.value;
        expandDrawerSearch(state.ui.phaseDrawerQuery);
        state.ui.phaseDrawerOpen = true;
        const caret = e.target.selectionStart;
        save(); render();
        const search = $('#phaseDrawerSearch');
        if (search) { try { search.focus({ preventScroll:true }); } catch (_) { search.focus(); } try { search.setSelectionRange(caret, caret); } catch (_) {} }
        return;
      }
      if (e.target.id === 'q') {
        state.ui.q = e.target.value; save();
        clearTimeout(window.__qT);
        window.__qT = setTimeout(() => {
          render(); const q = $('#q'); if (q) { q.focus(); q.setSelectionRange(q.value.length, q.value.length); }
        }, 320);
      }
    });

    app.addEventListener('click', (e) => {
      const btn = e.target.closest('[data-act]');
      if (!btn) return;
      const act = btn.dataset.act, id = btn.dataset.id;
      // کلیک روی checkbox فاز فقط وضعیت را تغییر می دهد، نه باز/بسته بودن آن را.
      if (act === 'toggle-phase' && (e.target.closest('[data-field]') || e.target.closest('.ph-ck'))) return;
      if (state.ui.editing && e.target.closest('[contenteditable="true"]') && act !== 'toggle-phase') return;

      switch (act) {
        case 'toggle-phase': {
          const ph = findPhase(id); if (!ph) return;
          state.open[id] = !state.open[id];
          save();
          const el = document.getElementById(id); if (el) el.classList.toggle('open', !!state.open[id]);
          break;
        }
        case 'side-toggle': {
          if (!state.ui.sideClosed) state.ui.sideClosed = { dev: true };
          state.ui.sideClosed[id] = !state.ui.sideClosed[id];
          save(); render();
          break;
        }
        case 'sidebar-layout-cycle': {
          const idx = SIDEBAR_LAYOUTS.indexOf(validSidebarLayout(state.ui.sidebarLayout));
          state.ui.sidebarLayout = SIDEBAR_LAYOUTS[(idx + 1) % SIDEBAR_LAYOUTS.length];
          saveNow(); render();
          toast('چیدمان سایدبار: ' + (state.ui.sidebarLayout === 'dual-right' ? 'دو سایدبار، فهرست فازها راست' : state.ui.sidebarLayout === 'single-right' ? 'یک سایدبار در سمت راست' : 'دو سایدبار، فهرست فازها چپ'));
          break;
        }
        case 'font-scale': adjustFontScale(btn.dataset.scope || 'ui', btn.dataset.dir); break;
        case 'phase-drawer-close': togglePhaseDrawer(false); break;
        case 'side-phase-toggle': {
          e.stopPropagation();
          if (!state.ui.sideOpen) state.ui.sideOpen = {};
          state.ui.sideOpen[id] = !state.ui.sideOpen[id];
          save(); render();
          break;
        }
        case 'side-step-toggle': {
          e.stopPropagation();
          if (!state.ui.sideStepOpen) state.ui.sideStepOpen = {};
          state.ui.sideStepOpen[id] = !state.ui.sideStepOpen[id];
          save(); render();
          break;
        }
        case 'goto-step': if (btn.closest('#phaseNavDrawer')) state.ui.phaseDrawerOpen = false; navigateToEntity('step', id); break;
        case 'goto-sub': if (btn.closest('#phaseNavDrawer')) state.ui.phaseDrawerOpen = false; navigateToEntity('sub', id); break;
        case 'goto': if (btn.closest('#phaseNavDrawer')) state.ui.phaseDrawerOpen = false; navigateToEntity('phase', id); break;
        case 'entity-menu': {
          e.stopPropagation();
          openEntityMenu(btn, btn.dataset.etype, btn.dataset.eid);
          break;
        }
        case 'mark-filter': {
          const v = btn.dataset.id;
          if (!Array.isArray(state.ui.markFilters)) state.ui.markFilters = [];
          const i = state.ui.markFilters.indexOf(v);
          if (i === -1) state.ui.markFilters.push(v); else state.ui.markFilters.splice(i, 1);
          save(); render();
          toast(state.ui.markFilters.length ? 'فیلتر نشان فعال شد: ' + state.ui.markFilters.join(' + ') : 'فیلتر نشان پاک شد.');
          break;
        }
        case 'mark-clear': state.ui.markFilters = []; save(); render(); break;
        case 'note-toggle': {
          const key = btn.dataset.key;
          state.openNotes[key] = !state.openNotes[key]; save();
          const wrap = document.querySelector('[data-note="' + key + '"]');
          if (wrap) {
            const w = wrap.closest('.note-wrap');
            w.classList.toggle('show', !!state.openNotes[key]);
            if (state.openNotes[key]) wrap.focus();
          }
          break;
        }
        case 'filter': state.ui.filter = id; save(); render(); break;
        case 'stack-filter': {
          const v = btn.dataset.id || '';
          state.ui.stack = (state.ui.stack === v) ? '' : v;
          save(); render();
          window.scrollTo(0, 0);
          if (state.ui.stack) toast('فیلتر استک: ' + state.ui.stack);
          break;
        }
        case 'stack-clear': state.ui.stack = ''; save(); render(); break;
        case 'expand-all': state.plan.forEach(p => state.open[p.id] = true); save(); render(); break;
        case 'collapse-all': state.open = {}; save(); render(); break;
        case 'edit-mode':
          state.ui.editing = !state.ui.editing;
          save(); render();
          toast(state.ui.editing
            ? 'حالت ویرایش روشن شد — روی هر متن کلیک کن، یا با «باز کردن همه» دکمه های افزودن/جابه جایی را ببین.'
            : 'حالت ویرایش خاموش شد — تغییرات ذخیره شد ✍️');
          break;
        case 'export': exportJSON(); break;
        case 'export-answers': exportAnswers(); break;
        case 'import': importJSON(); break;
        case 'report': reportMD(btn.dataset.id); break;
        case 'report-menu': {
          const w = btn.closest('.menu-wrap');
          const openNow = w.classList.contains('open');
          closeMenus();
          if (!openNow) { updateStats(); w.classList.add('open'); }
          break;
        }
        case 'notebook': nbToggle(); break;
        case 'notebook-copy': nbCopy(); break;
        case 'notebook-share': nbShare(); break;
        case 'clean-all': cleanAll(); break;
        case 'file-connect': fileStore.connect(); break;
        case 'file-regrant': fileStore.regrant(); break;
        case 'file-write': writeSidecarNow(); break;
        case 'file-disconnect': fileStore.disconnect(); break;
        case 'copy-json': copyJSON(); break;
        case 'print': window.print(); break;
        case 'move-phase': movePhaseTo(id, btn.dataset.dir); break;
        case 'add-step': addStep(id, null, btn.closest('.insert-action-group')?.querySelector('[data-insert-position]')?.value); break;
        case 'add-phase': addPhase(id, btn.closest('.insert-action-group')?.querySelector('[data-insert-position]')?.value); break;
        case 'del-phase': delPhase(id); break;
        case 'del-step': delStep(id); break;
        case 'add-sub': addSub(id, btn.dataset.after, btn.closest('.insert-action-group')?.querySelector('[data-insert-position]')?.value); break;
        case 'del-sub': delSub(id, btn.dataset.sub); break;
        case 'move-step': moveStep(id, btn.dataset.dir); break;
        case 'move-sub': moveSub(id, btn.dataset.sub, btn.dataset.dir); break;
        case 'add-tip': {
          const f = findStep(id); if (!f) return;
          f.s.tip = f.s.tip || 'نکته تجربی: …'; touch(); render();
          const el = document.getElementById(id);
          if (el) { const t = el.querySelector('.tip [data-field="tip"]'); if (t) t.focus(); }
          break;
        }

        case 'reset-content':
          if (confirm('ساختار مراحل به حالت پیش فرض این فایل بازگردد؟ (تیک ها حفظ می شوند)')) {
            state.plan = clone(DEFAULT_PLAN); state.customized = false; state.ui.stale = false;
            saveNow(); render(); toast('محتوای پیش فرض بازنشانی شد');
          }
          break;
        case 'apply-stale':
          state.plan = clone(DEFAULT_PLAN); state.customized = false; state.ui.stale = false;
          saveNow(); render(); toast('محتوای جدید بارگذاری شد');
          break;
        case 'dismiss-stale': state.ui.stale = false; saveNow(); $('#banner').classList.remove('show'); break;
      }
    });

    // بعد از پایان یادداشت: به روزرسانی نشانگرهای همان مرحله
    app.addEventListener('focusout', (e) => {
      const n = e.target.closest('textarea[data-note]');
      if (n) {
        const id = n.dataset.note.replace(/^n:/, '');
        let fs = findStep(id), ph = null, st = null;
        if (fs) { ph = fs.ph; st = fs.s; }
        else { const o = findStepOfSub(id); if (o) { ph = o.ph; st = o.s; } }
        if (st) {
          const el = document.getElementById(st.id);
          if (el) el.outerHTML = stepHTML(ph, st, ph.steps.indexOf(st));
          updateStats(); save();
        }
        return;
      }
      const el2 = e.target.closest('[data-field]');
      if (el2 && state.ui.editing) commitField(el2);
    });

    app.addEventListener('keydown', (e) => {
      const el = e.target.closest('[contenteditable="true"]');
      if (el && e.key === 'Enter' && !e.shiftKey) { e.preventDefault(); el.blur(); }
    });

    // دکمه شناور، Drawer فهرست فازها را از سمت چپ باز و بسته می کند.
    const phaseFab = document.getElementById('phaseFab');
    if (phaseFab) phaseFab.addEventListener('click', () => togglePhaseDrawer());

    // دکمه بازگشت به بالا (بیرون از #app است)
    const toTop = document.getElementById('totop');
    if (toTop) {
      toTop.addEventListener('click', () => window.scrollTo({ top: 0, behavior: 'smooth' }));
      let ticking = false;
      window.addEventListener('scroll', () => {
        if (ticking) return; ticking = true;
        requestAnimationFrame(() => {
          toTop.classList.toggle('show', window.scrollY > 700);
          updateHeaderCompact();
          ticking = false;
        });
      }, { passive: true });
    }

    // میان برهای اختصاصی بدون Alt و دور از میان برهای رایج مرورگر.
    // Ctrl+Shift+. = خروجی JSON | Ctrl+Shift+; = ساخت یادداشت جدید
    document.addEventListener('keydown', (e) => {
      const tag = (e.target.tagName || '').toLowerCase();
      const typing = tag === 'input' || tag === 'textarea' || e.target.isContentEditable;
      const mod = e.ctrlKey || e.metaKey;
      if (mod && e.shiftKey && !e.altKey && !typing && e.code === 'Period') { e.preventDefault(); exportJSON(); return; }
      if (mod && e.shiftKey && !e.altKey && !typing && e.code === 'Semicolon') {
        e.preventDefault();
        if (!state.ui.nbOpen) nbToggle(true);
        setTimeout(() => nbCreate('primary'), state.ui.nbOpen ? 0 : 120);
        return;
      }
      if (e.key === '/' && !typing) { e.preventDefault(); const q = $('#q'); if (q) q.focus(); }
    });
  }

  /* ---------- اسکرول اسپای + اسکرول مستقل لیست فازها -------------------
   * دو مکانیزم اسکرول:
   *  ۱) مکانیزم اصلی: اسکرول خود صفحه برای Phases / Steps / Sub-steps
   *  ۲) مکانیزم سایدبار: aside.side دارای overflow مستقل است و لیست فازها
   *     (.navlist) هم اسکرول داخلی خودش را دارد؛ با اسکرول صفحه، آیتم فعال
   *     خودکار داخل لیست دیده می شود.
   * ------------------------------------------------------------------- */
  let spyObs = null;
  function scrollNavTo(id) {
    const list = $('.navlist'); if (!list) return;
    const item = list.querySelector('[data-act="goto"][data-id="' + id + '"]'); if (!item) return;
    const top = item.offsetTop, h = item.offsetHeight;
    if (top < list.scrollTop + 6) list.scrollTop = Math.max(0, top - 10);
    else if (top + h > list.scrollTop + list.clientHeight - 6) list.scrollTop = top + h - list.clientHeight + 10;
  }
  function spy() {
    if (spyObs) spyObs.disconnect();
    if (!('IntersectionObserver' in window)) return;
    spyObs = new IntersectionObserver((ents) => {
      ents.forEach(en => {
        if (en.isIntersecting) {
          $$('.navitem').forEach(n => n.classList.toggle('on', n.dataset.id === en.target.id));
          scrollNavTo(en.target.id);
        }
      });
    }, { rootMargin: '-90px 0px -70% 0px', threshold: 0 });
    $$('.phase').forEach(p => spyObs.observe(p));
  }

  // پیش فرض: دو فاز اول باز (سایر فازها با کلیک/جست وجو باز می شوند تا صفحه سبک بماند)
  if (!Object.keys(state.open).length) state.plan.slice(0, 2).forEach(p => { state.open[p.id] = true; });
  let booted = false;
  function boot() {
    if (booted) return; booted = true;
    nbMount();
    entMenuMount();
    bind();
    render();
    nbRender();
    // Run immediately for deep links, then re-run only if sidecar/handle loading replaced state.
    handleDeepLink();
    bootStores().then(used => { if (used) handleDeepLink(); });
    window.addEventListener('hashchange', handleDeepLink);
  }
  async function bootStores() {
    let used = false;
    try { used = await fileStore.restoreFromIdb(); } catch (e) { }
    if (!used && !fileStore.connected) {
      const side = await fileStore.fetchSidecar();
      if (side && !answersEmpty(side)) {
        const localEmpty = answersEmpty(state);
        const newer = side.savedAt && state.savedAt ? (Date.parse(side.savedAt) > Date.parse(state.savedAt)) : false;
        if (localEmpty || newer) { applyAnswers(side, true); used = true; }
      }
    }
    if (used) {
      saveNow(); render(); nbRender();
      toast('پاسخ ها از ' + SIDECAR + ' بارگذاری شد ✅');
    } else if (fileStore.status !== 'connected') {
      render(); nbRender();
    }
    return used;
  }
  document.addEventListener('DOMContentLoaded', boot);
  if (document.readyState !== 'loading') boot();
})();
